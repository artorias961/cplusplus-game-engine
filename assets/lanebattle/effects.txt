# ---------------------------------------------------------------------------
# effects.txt - how MUCH of every effect there is, how OFTEN it appears, and
# how BIG it is. Edit and run the game; no compiler involved.
#
# Nothing here changes a battle. Effects, weather and ambience are pictures:
# they spawn from their own dice, never from anything a fight reads, and a
# missing or broken file just leaves the compiled-in defaults (Art.cpp).
#
# THREE SIZES. Everything that appears in numbers - a raindrop, a cloud, a
# sword arc - is spawned SMALL, MEDIUM or LARGE, so a crowd of them reads as
# depth and variety rather than a grid of stamps. Each entry says how big its
# MEDIUM one is and what mix of the three it spawns; [sizes] says how much
# smaller a small one is and how much bigger a large one.
#
#   small / medium / large   of every hundred spawned, how many of each.
#                            They need not add up to a hundred; they are
#                            weights. "0 0 100" spawns only large ones.
#
# LIMITS are hard. However hard it rains or however big the melee, there are
# never more than `limit` of a thing on screen at once - spawning simply
# waits. `spawn` is how fast new ones arrive while there is room, so a low
# spawn with a high limit makes weather that builds up, and a high spawn with
# a low limit makes it arrive at once and then hold steady.
# ---------------------------------------------------------------------------

[sizes]
small  = 0.6        # a SMALL one is 0.6 times a medium one
medium = 1.0
large  = 1.5        # a LARGE one is half again as big
